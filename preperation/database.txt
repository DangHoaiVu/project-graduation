-- WARNING: This schema is for context only and is not meant to be run.
-- Table order and constraints may not be valid for execution.

CREATE TABLE public.users (
  moodle_user_id integer NOT NULL,
  role character varying NOT NULL,
  name character varying,
  CONSTRAINT users_pkey PRIMARY KEY (moodle_user_id)
);
CREATE TABLE public.fcm_tokens (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  user_id integer,
  token text NOT NULL UNIQUE,
  device_type character varying,
  last_used_at timestamp with time zone DEFAULT now(),
  device_id character varying NOT NULL,
  CONSTRAINT fcm_tokens_pkey PRIMARY KEY (id),
  CONSTRAINT fcm_tokens_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(moodle_user_id)
);
CREATE TABLE public.moodle_credentials (
  user_id integer NOT NULL,
  encrypted_token text NOT NULL,
  updated_at timestamp with time zone NOT NULL DEFAULT now(),
  CONSTRAINT moodle_credentials_pkey PRIMARY KEY (user_id),
  CONSTRAINT moodle_credentials_user_id_users_moodle_user_id_fk FOREIGN KEY (user_id) REFERENCES public.users(moodle_user_id)
);
CREATE TABLE public.notification_deliveries (
  user_id integer NOT NULL,
  event_id integer NOT NULL,
  event_type character varying NOT NULL,
  reminder_minutes integer NOT NULL,
  event_timestamp timestamp with time zone NOT NULL,
  sent_at timestamp with time zone NOT NULL DEFAULT now(),
  CONSTRAINT notification_deliveries_pkey PRIMARY KEY (user_id, event_id, event_type, reminder_minutes),
  CONSTRAINT notification_deliveries_user_id_users_moodle_user_id_fk FOREIGN KEY (user_id) REFERENCES public.users(moodle_user_id)
);
CREATE TABLE public.moodle_events (
  user_id integer NOT NULL,
  event_id integer NOT NULL,
  event_type character varying NOT NULL,
  title character varying NOT NULL,
  timestart integer NOT NULL,
  CONSTRAINT moodle_events_pkey PRIMARY KEY (user_id, event_id, event_type)
);
CREATE TABLE public.push_queue (
  id integer NOT NULL DEFAULT nextval('push_queue_id_seq'::regclass),
  user_id integer NOT NULL,
  event_id integer NOT NULL,
  event_type character varying NOT NULL,
  title character varying NOT NULL,
  reminder_minutes integer NOT NULL,
  created_at timestamp with time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT push_queue_pkey PRIMARY KEY (id)
);
